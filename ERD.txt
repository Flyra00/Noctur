////////////////////////////////////////////////////////////
// CITY DISCOVERY PLATFORM
// ERD / DATABASE DESIGN
//
// Stack:
// - PostgreSQL
// - PostGIS
// - Prisma
// - Next.js
//
// Core concept:
// User -> Place/Business/Event/Community/Journey
// User interactions -> Save/Review/RSVP/Check-in/Follow
// Operations -> Claim/Verification/Moderation/Audit
////////////////////////////////////////////////////////////


////////////////////////////////////////////////////////////
// ENUM-LIKE LOOKUP TABLES / RBAC
////////////////////////////////////////////////////////////

Table roles {
  id uuid [pk]
  name varchar(50) [unique, not null]
  description text
  created_at timestamp
  updated_at timestamp
}

Table permissions {
  id uuid [pk]
  name varchar(100) [unique, not null]
  description text
  created_at timestamp
  updated_at timestamp
}

Table user_roles {
  user_id uuid [not null]
  role_id uuid [not null]
  created_at timestamp

  indexes {
    (user_id, role_id) [pk]
  }
}

Table role_permissions {
  role_id uuid [not null]
  permission_id uuid [not null]
  created_at timestamp

  indexes {
    (role_id, permission_id) [pk]
  }
}


////////////////////////////////////////////////////////////
// AUTHENTICATION / USERS
////////////////////////////////////////////////////////////

Table users {
  id uuid [pk]

  name varchar(150) [not null]
  username varchar(50) [unique]
  email varchar(255) [unique, not null]
  email_verified_at timestamp
  password_hash varchar(255)

  image_url text
  bio text

  city_id uuid

  status varchar(30) [not null, note: 'ACTIVE | SUSPENDED | DELETED']

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp

  indexes {
    email
    username
    city_id
    status
  }
}


// Auth.js / OAuth provider accounts
Table accounts {
  id uuid [pk]
  user_id uuid [not null]

  type varchar(50) [not null]
  provider varchar(100) [not null]
  provider_account_id varchar(255) [not null]

  refresh_token text
  access_token text
  expires_at bigint
  token_type varchar(100)
  scope text
  id_token text
  session_state text

  created_at timestamp
  updated_at timestamp

  indexes {
    (provider, provider_account_id) [unique]
    user_id
  }
}


// Auth.js sessions
Table sessions {
  id uuid [pk]
  session_token varchar(255) [unique, not null]
  user_id uuid [not null]
  expires_at timestamp [not null]

  created_at timestamp
  updated_at timestamp

  indexes {
    user_id
  }
}


// Auth.js verification tokens
Table verification_tokens {
  identifier varchar(255) [not null]
  token varchar(255) [not null]
  expires_at timestamp [not null]

  indexes {
    (identifier, token) [pk]
  }
}

// Physical location.
// A Business represents the operating entity.
// A Place represents its physical branch/location.
Table places {
  id uuid [pk]

  business_id uuid
  city_id uuid

  name varchar(200) [not null]

  description text

  address text
  postal_code varchar(20)

  latitude decimal(10,7)
  longitude decimal(10,7)

  // PostgreSQL/PostGIS
  location geography

  phone varchar(50)
  website_url text

  status varchar(30) [not null, note: 'DRAFT | PUBLISHED | HIDDEN | CLOSED']

  price_level varchar(20, note: 'FREE | LOW | MEDIUM | HIGH | LUXURY')

  created_by uuid
  updated_by uuid

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp

  indexes {
    slug
    business_id
    city_id
    category_id
    status
    (latitude, longitude)
  }
}

  indexes {

////////////////////////////////////////////////////////////
// RELATIONSHIPS
////////////////////////////////////////////////////////////


// RBAC
Ref: user_roles.user_id > users.id
Ref: user_roles.role_id > roles.id

Ref: role_permissions.role_id > roles.id
Ref: role_permissions.permission_id > permissions.id


// Authentication
Ref: accounts.user_id > users.id
Ref: sessions.user_id > users.id


// Geography
Ref: users.city_id > cities.id

Ref: categories.parent_id > categories.id


// Business
Ref: business_members.business_id > businesses.id
Ref: business_members.user_id > users.id

Ref: business_claims.business_id > businesses.id
Ref: business_claims.submitted_by > users.id
Ref: business_claims.reviewed_by > users.id

Ref: business_verifications.business_id > businesses.id
Ref: business_verifications.claim_id > business_claims.id
Ref: business_verifications.verified_by > users.id


// Places
Ref: places.business_id > businesses.id
Ref: places.city_id > cities.id
Ref: places.category_id > categories.id
